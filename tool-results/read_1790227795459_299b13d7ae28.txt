    64→/** Fetch records by exact slug list (order preserved by the API). */
    65→function useSlugFetch(slugs: string, enabled = true): SourceRecord[] | null {
    66→  const [records, setRecords] = React.useState<SourceRecord[] | null>(null);
    67→  React.useEffect(() => {
    68→    if (!enabled) return;
    69→    let alive = true;
    70→    setRecords(null);
    71→    fetch(`/api/content?slugs=${encodeURIComponent(slugs)}`)
    72→      .then((r) => (r.ok ? r.json() : Promise.reject(new Error('load failed'))))
    73→      .then((d: { items?: SourceRecord[] }) => {
    74→        if (alive) setRecords(d.items ?? []);
    75→      })
    76→      .catch(() => {
    77→        if (alive) setRecords([]);
    78→      });
    79→    return () => {
    80→      alive = false;
    81→    };
    82→  }, [slugs, enabled]);
    83→  return records;
    84→}
    85→
    86→function findBySlug(records: SourceRecord[] | null, slug: string): SourceRecord | null {
    87→  return records?.find((r) => r.slug === slug) ?? null;

   185→
   186→  const lat = profile?.locationLat ?? DEFAULT_LAT;
   187→  const lng = profile?.locationLng ?? DEFAULT_LNG;
   188→  const methodKey = profile?.prayerMethod ?? 'MWL';
   189→  const asrFactor = (profile?.asrFactor === 2 ? 2 : 1) as 1 | 2;

   192→
   193→  React.useEffect(() => {
   194→    setMounted(true);
   195→    // Battery-conscious ticking: pause while the tab is hidden, resync on return.
   196→    const tick = () => setNow(new Date());

   198→    const start = () => {

   200→    };

   203→        clearInterval(t);

   211→        start();

   213→    };

   215→    document.addEventListener('visibilitychange', onVisibility);

   249→
   250→  const onMethodChange = async (v: string) => {

   258→
   259→  const onAsrChange = (factor: 1 | 2) => {

   264→  };
   265→
   266→  const locate = () => {
   267→    if (typeof navigator === 'undefined' || !('geolocation' in navigator)) {
   268→      toast({
   269→        title: 'Location unavailable',
   270→        description: 'This browser does not expose geolocation.',
   271→        variant: 'destructive',
   272→      });
   273→      return;
   274→    }
   275→    setLocating(true);
