    88→    88→async function fetchJson<T>(url: string, init?: RequestInit): Promise<T | null> {
    89→    89→  try {
    90→    90→    const res = await fetch(url, { cache: 'no-store', ...init });
    91→    91→    if (!res.ok) return null;
    92→    92→    return (await res.json()) as T;
    93→    93→  } catch {
    94→    94→    return null;
    95→    95→  }
    96→    96→}

   255→   255→function SurahReader({
   256→   256→  surahNumber,
   257→   257→  lastRead,
   258→   258→  onBack,
   259→   259→  onProgress,
   260→   260→}: {
   261→   261→  surahNumber: number;
   262→   262→  lastRead?: number;
   263→   263→  onBack: () => void;
   264→   264→  onProgress: (surahNumber: number, lastAyah: number) => void;
   265→   265→}) {
   266→   266→  const { toast } = useToast();
   267→   267→  const toggleBookmark = useApp((s) => s.toggleBookmark);
   268→   268→  const openRecordBySlug = useApp((s) => s.openRecordBySlug);
   269→   269→  const bookmarkSlugs = useApp((s) => s.bookmarkSlugs);
   270→   270→
   271→   271→  const [data, setData] = React.useState<SurahResponse | null>(null);
   272→   272→  const [state, setState] = React.useState<'loading' | 'ready' | 'error'>('loading');
   273→   273→  const [reloadTick, setReloadTick] = React.useState(0);
   274→   274→  const [marking, setMarking] = React.useState<number | null>(null);
   275→   275→
   276→   276→  React.useEffect(() => {
   277→   277→    let alive = true;
   278→   278→    setState('loading');
   279→   279→    void (async () => {
   280→   280→      const res = await fetchJson<SurahResponse>(`/api/quran/surah/${surahNumber}`);
   281→   281→      if (!alive) return;
   282→   282→      if (res && Array.isArray(res.ayahs)) {
   283→   283→        setData(res);
   284→   284→        setState('ready');
   285→   285→      } else {
   286→   286→        setState('error');
   287→   287→      }
   288→   288→    })();
   289→   289→    return () => {
   290→   290→      alive = false;
   291→   291→    };
   292→   292→  }, [surahNumber, reloadTick]);
   293→   293→
   294→   294→  // After loading, jump to the last-read ayah (or the top of the surah header).
   295→   295→  React.useEffect(() => {
   296→   296→    if (state !== 'ready') return;
   297→   297→    const lastReadEl = typeof lastRead === 'number' ? document.getElementById(`ayah-${surahNumber}-${lastRead}`) : null;
   298→   298→    const target = lastReadEl ?? document.getElementById(`surah-top-${surahNumber}`);
   299→   299→    if (target) {
   300→   300→      requestAnimationFrame(() => {
   301→   301→        target.scrollIntoView({ behavior: 'smooth', block: lastReadEl ? 'center' : 'start' });
   302→   302→      });
   303→   303→    }
   304→   304→  }, [state, surahNumber]);
   305→   305→
   306→   306→  const markLastRead = async (ayah: number) => {
   307→   307→    if (marking !== null) return;
   308→   308→    setMarking(ayah);
   309→   309→    const res = await fetchJson<{ ok: boolean }>('/api/progress', {
   310→   310→      method: 'POST',
   311→   311→      headers: { 'Content-Type': 'application/json' },
   312→   312→      body: JSON.stringify({ kind: 'reading', surahNumber, lastAyah: ayah }),
   313→   313→    });
   314→   314→    setMarking(null);
   315→   315→    if (res?.ok) {
   316→   316→      onProgress(surahNumber, ayah);
   317→   317→      toast({
   318→   318→        title: 'Reading position saved',
   319→   319→        description: `Last read: ${getSurah(surahNumber)?.transliteration ?? `Surah ${surahNumber}`} · ayah ${ayah}.`,
   320→   320→      });
   321→   321→    } else {
   322→   322→      toast({
   323→   323→        title: 'Could not save reading position',
   324→   324→        description: 'Please check your connection and try again.',
   325→   325→      });
   326→   326→    }
   327→   327→  };
   328→   328→
   329→   329→  const surahMeta = data?.surah ?? getSurah(surahNumber);
   330→   330→
   331→   331→  return (
   332→   332→    <section aria-label={`Surah ${surahNumber} reader`}>
   333→   333→      {/* Reader toolbar */}
   334→   334→      <div className="flex items-center justify-between gap-2 mb-4">
   335→   335→        <Button
   336→   336→          variant="ghost"
   337→   337→          onClick={onBack}
   338→   338→          className="h-11 px-3 -ml-2 text-muted-foreground hover:text-foreground"
   339→   339→          aria-label="Back to the surah list"
   340→   340→        >
   341→   341→          <ChevronLeft className="h-5 w-5 mr-1" aria-hidden />
   342→   342→          All surahs
   343→   343→        </Button>
   344→   344→        {typeof lastRead === 'number' && (
   345→   345→          <span className="inline-flex items-center gap-1.5 text-xs text-muted-foreground bg-muted/60 border border-border/60 rounded-full px-3 py-2">
   346→   346→            <BookmarkCheck className="h-3.5 w-3.5 text-primary" aria-hidden />
   347→   347→            Last read: ayah {lastRead}
   348→   348→          </span>
   349→   349→        )}
   350→   350→      </div>
   351→   351→
   352→   352→      {state === 'loading' && (
   353→   353→        <div role="status" aria-busy="true" className="space-y-4">
   354→   354→          <span className="sr-only">Loading surah…</span>
   355→   355→          <Skeleton className="h-44 w-full rounded-xl" />
   356→   356→          <Skeleton className="h-36 w-full rounded-xl" />
   357→   357→          <Skeleton className="h-36 w-full rounded-xl" />
   358→   358→        </div>
   359→   359→      )}
   360→   360→
   361→   361→      {state === 'error' && (
   362→   362→        <ErrorState what={`surah ${surahNumber}`} onRetry={() => setReloadTick((t) => t + 1)} />
   363→   363→      )}
   364→   364→
   365→   365→      {state === 'ready' && data && surahMeta && (
   366→   366→        <>
   367→   367→          {/* Surah header card */}
   368→   368→          <Card className="paper-card overflow-hidden border-border/80 mb-5">
   369→   369→            <div className="pattern-khatim pattern-fade h-1.5 w-full" aria-hidden />
   370→   370→            <CardContent className="p-4 sm:p-6 text-center">
   371→   371→              <p className="text-[0.65rem] uppercase tracking-[0.25em] text-muted-foreground">
   372→   372→                Surah {surahMeta.number} of 114
   373→   373→              </p>
   374→   374→              <h2
   375→   375→                id={`surah-top-${surahNumber}`}
   376→   376→                className="font-arabic text-4xl sm:text-5xl text-foreground leading-[1.7] my-3"
   377→   377→                dir="rtl"
   378→   378→                lang="ar"
   379→   379→              >
   380→   380→                {surahMeta.arabic}
   381→   381→              </h2>
   382→   382→              <p className="font-display text-xl sm:text-2xl font-semibold text-foreground">{surahMeta.transliteration}</p>
   383→   383→              <p className="text-sm text-muted-foreground italic mt-0.5">{surahMeta.english}</p>
   384→   384→              <div className="flex items-center justify-center gap-2 flex-wrap mt-4">
   385→   385→                <Badge variant="outline" className="rounded-full px-3 py-1 text-xs font-medium">
   386→   386→                  {surahMeta.place}
   387→   387→                </Badge>
   388→   388→                <Badge variant="outline" className="rounded-full px-3 py-1 text-xs font-medium">
   389→   389→                  {surahMeta.ayahs} {surahMeta.ayahs === 1 ? 'ayah' : 'ayahs'}
   390→   390→                </Badge>
   391→   391→                <Badge
   392→   392→                  variant="outline"
   393→   393→                  className={cn(
   394→   394→                    'rounded-full px-3 py-1 text-xs font-medium',
   395→   395→                    data.status === 'local-curated' ? 'border-gold/50 text-gold' : 'border-primary/30 text-primary'
   396→   396→                  )}
   397→   397→                >
   398→   398→                  {STATUS_LABEL[data.status]}
   399→   399→                </Badge>
   400→   400→              </div>
   401→   401→              <p className="text-[0.68rem] text-muted-foreground mt-4 leading-relaxed">
   402→   402→                Translation: {data.translationSource}
   403→   403→                {data.source ? ` · ${data.source}` : ''}
   404→   404→              </p>
   405→   405→            </CardContent>
   406→   406→          </Card>
   407→   407→
   408→   408→          {data.status === 'local-curated' && (
   409→   409→            <Alert className="mb-5 border-gold/40 bg-gold/10">
   410→   410→              <CloudOff className="h-4 w-4 text-gold" aria-hidden />
   411→   411→              <AlertTitle className="text-foreground">Only BASIRA&apos;s curated ayahs are available right now</AlertTitle>
   412→   412→              <AlertDescription>{data.note}</AlertDescription>
   413→   413→            </Alert>
   414→   414→          )}
   415→   415→
   416→   416→          {data.ayahs.length === 0 ? (
   417→   417→            <EmptyState
   418→   418→              icon={<BookOpen className="h-10 w-10" aria-hidden />}
   419→   419→              title="No curated ayahs for this surah yet"
   420→   420→              hint="The external Quran dataset could not be reached and BASIRA has not curated this surah yet. Try the Topics tab — every ayah there is verified."
   421→   421→            />
   422→   422→          ) : (
   423→   423→            <div className="space-y-3 sm:space-y-4">
   424→   424→              {data.ayahs.map((a) => {
   425→   425→                const isLastRead = lastRead === a.numberInSurah;
   426→   426→                const bookmarked = a.slug ? bookmarkSlugs.has(a.slug) : false;
   427→   427→                return (
   428→   428→                  <article
   429→   429→                    key={a.numberInSurah}
   430→   430→                    id={`ayah-${surahNumber}-${a.numberInSurah}`}
   431→   431→                    className={cn(
   432→   432→                      'paper-card rounded-xl border p-4 sm:p-5 transition-shadow',
   433→   433→                      isLastRead ? 'border-primary/50 shadow-md' : 'border-border/80 shadow-sm hover:shadow-md'
   434→   434→                    )}
   435→   435→                  >
   436→   436→                    <div className="flex items-center justify-between gap-2">
