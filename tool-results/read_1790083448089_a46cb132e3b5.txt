    70→    70→// ————————————————— markdown-lite answer rendering —————————————————
    71→    71→// Supports: ### h3, #### h4, - bullets, 1. numbered items, **bold**,
    72→    72→// and [S#] citation tokens rendered as inspectable CitationChips.
    73→    73→
    74→    74→type Block =
    75→    75→  | { kind: 'h3' | 'h4' | 'p'; text: string }
    76→    76→  | { kind: 'ul'; items: string[] }
    77→    77→  | { kind: 'ol'; items: { num: number; text: string }[] }
    78→    78→  | { kind: 'quote'; lines: string[] };
    79→    79→
    80→    80→function parseBlocks(answer: string): Block[] {
    81→    81→  const blocks: Block[] = [];
    82→    82→  let list: Extract<Block, { kind: 'ul' | 'ol' }> | null = null;
    83→    83→  let quote: Extract<Block, { kind: 'quote' }> | null = null;
    84→    84→  const flush = () => {
    85→    85→    if (list) {
    86→    86→      blocks.push(list);
    87→    87→      list = null;
    88→    88→    }
    89→    89→    if (quote) {
    90→    90→      blocks.push(quote);
    91→    91→      quote = null;
    92→    92→    }
    93→    93→  };
    94→    94→  for (const rawLine of answer.split('\n')) {
    95→    95→    const line = rawLine.trim();
    96→    96→    if (!line) continue; // blank lines do not split a list apart
    97→    97→    if (line.startsWith('#### ')) {
    98→    98→      flush();
    99→    99→      blocks.push({ kind: 'h4', text: line.slice(5) });
   100→   100→      continue;
   101→   101→    }
   102→   102→    if (line.startsWith('### ')) {
   103→   103→      flush();
   104→   104→      blocks.push({ kind: 'h3', text: line.slice(4) });
   105→   105→      continue;
   106→   106→    }
   107→   107→    if (line.startsWith('>')) {
   108→   108→      if (!quote) {
   109→   109→        flush();
   110→   110→        quote = { kind: 'quote', lines: [] };
   111→   111→      }
   112→   112→      quote.lines.push(line.replace(/^>\s?/, ''));
   113→   113→      continue;
   114→   114→    }
   115→   115→    if (line.startsWith('- ')) {
   116→   116→      if (quote) {
   117→   117→        blocks.push(quote);
   118→   118→        quote = null;
   119→   119→      }
   120→   120→      if (!list || list.kind !== 'ul') {
   121→   121→        flush();
   122→   122→        list = { kind: 'ul', items: [] };
   123→   123→      }
   124→   124→      list.items.push(line.slice(2));
   125→   125→      continue;
   126→   126→    }
   127→   127→    const ordered = line.match(/^(\d+)\. +(.*)$/);
   128→   128→    if (ordered) {
   129→   129→      if (quote) {
   130→   130→        blocks.push(quote);
   131→   131→        quote = null;
   132→   132→      }
   133→   133→      if (!list || list.kind !== 'ol') {
   134→   134→        flush();
   135→   135→        list = { kind: 'ol', items: [] };
   136→   136→      }
   137→   137→      list.items.push({ num: Number(ordered[1]), text: ordered[2] });
   138→   138→      continue;
   139→   139→    }
   140→   140→    flush();
   141→   141→    blocks.push({ kind: 'p', text: line });
   142→   142→  }
   143→   143→  flush();
   144→   144→  return blocks;
   145→   145→}

   147→   147→/** Tokenize a single line into text / <strong> / <em> / CitationChip pieces. */
   148→   148→function renderInline(
   149→   149→  text: string,
   150→   150→  sources: SourceRecord[],
   151→   151→  openRecord: (record: SourceRecord) => void
   152→   152→): React.ReactNode[] {
   153→   153→  const nodes: React.ReactNode[] = [];
   154→   154→  const re = /(\*\*[^*]+\*\*)|(\*[^*]+\*)|\[S(\d+)\]/g;
   155→   155→  let cursor = 0;
   156→   156→  let key = 0;
   157→   157→  let m: RegExpExecArray | null;
   158→   158→  while ((m = re.exec(text)) !== null) {
   159→   159→    if (m.index > cursor) nodes.push(text.slice(cursor, m.index));
   160→   160→    if (m[1]) {
   161→   161→      nodes.push(
   162→   162→        <strong key={key++} className="font-semibold text-foreground">
   163→   163→          {m[1].slice(2, -2)}
   164→   164→        </strong>
   165→   165→      );
   166→   166→    } else if (m[2]) {
   167→   167→      nodes.push(
   168→   168→        <em key={key++} className="italic">
   169→   169→          {m[2].slice(1, -1)}
   170→   170→        </em>
   171→   171→      );
   172→   172→    } else if (m[3]) {
   173→   173→      const n = Number(m[3]);
   174→   174→      const record = sources[n - 1]; // sources[i] corresponds to [S{i+1}]
   175→   175→      nodes.push(
   176→   176→        <CitationChip
   177→   177→          key={key++}
   178→   178→          index={n}
   179→   179→          record={record}
   180→   180→          onClick={(r) => {
   181→   181→            if (r) openRecord(r);
   182→   182→          }}
   183→   183→        />
   184→   184→      );
   185→   185→    }
   186→   186→    cursor = m.index + m[0].length;
   187→   187→  }
   188→   188→  if (cursor < text.length) nodes.push(text.slice(cursor));
   189→   189→  return nodes;
   190→   190→}

   192→   192→function AnswerBody({ answer, sources }: { answer: string; sources: SourceRecord[] }) {
   193→   193→  const openRecord = useApp((s) => s.openRecord);
   194→   194→  const blocks = React.useMemo(() => parseBlocks(answer), [answer]);
   195→   195→  const inline = React.useCallback(
   196→   196→    (text: string) => renderInline(text, sources, openRecord),
   197→   197→    [sources, openRecord]
   198→   198→  );
   199→   199→  return (
   200→   200→    <div className="space-y-3">
   201→   201→      {blocks.map((b, i) => {
   202→   202→        if (b.kind === 'h3') {
   203→   203→          return (
   204→   204→            <h3 key={i} className="pt-1 font-display text-lg font-semibold leading-snug text-foreground">
   205→   205→              {inline(b.text)}
   206→   206→            </h3>
   207→   207→          );
   208→   208→        }
   209→   209→        if (b.kind === 'h4') {
   210→   210→          return (
   211→   211→            <h4 key={i} className="font-display text-[0.95rem] font-semibold leading-snug text-foreground/90">
   212→   212→              {inline(b.text)}
   213→   213→            </h4>
   214→   214→          );
   215→   215→        }
   216→   216→        if (b.kind === 'quote') {
   217→   217→          return (

   219→   219→              key={i}
   220→   220→              className="space-y-2 rounded-r-lg border-l-2 border-gold/60 bg-gold/5 py-2.5 pl-4 pr-3"

   222→   222→              {b.lines.map((line, j) => (
   223→   223→                <p key={j} className="text-[0.9rem] italic leading-relaxed text-foreground/85">

   227→   227→            </blockquote>

   233→   233→              {b.items.map((item, j) => (

   236→   236→                  <span>{inline(item)}</span>
   237→   237→                </li>
   238→   238→              ))}
   239→   239→            </ul>
   240→   240→          );
   241→   241→        }
   242→   242→        if (b.kind === 'ol') {
   243→   243→          return (
   244→   244→            <ol key={i} className="list-none space-y-1.5">
   245→   245→              {b.items.map((item, j) => (
   246→   246→                <li key={j} className="flex gap-2.5 leading-relaxed text-foreground/90">
   247→   247→                  <span
   248→   248→                    className="mt-0.5 flex h-5 min-w-5 shrink-0 items-center justify-center rounded-full bg-primary/10 text-[0.62rem] font-semibold tabular-nums text-primary"
   249→   249→                    aria-hidden
   250→   250→                  >
   251→   251→                    {item.num}
   252→   252→                  </span>
   253→   253→                  <span>{inline(item.text)}</span>
   254→   254→                </li>
   255→   255→              ))}
   256→   256→            </ol>
   257→   257→          );
   258→   258→        }
   259→   259→        return (
   260→   260→          <p key={i} className="leading-relaxed text-foreground/90">
   261→   261→            {inline(b.text)}
   262→   262→          </p>
   263→   263→        );
   264→   264→      })}
   265→   265→    </div>
   266→   266→  );

   470→   470→export function AskView() {
   471→   471→  const viewParams = useApp((s) => s.viewParams);
   472→   472→  const profile = useApp((s) => s.profile);
   473→   473→  const updateProfile = useApp((s) => s.updateProfile);
   474→   474→  const { toast } = useToast();
   475→   475→
   476→   476→  const restored = React.useMemo(restoreConversation, []);
   477→   477→  const [turns, setTurns] = React.useState<Turn[]>(restored.turns);
   478→   478→  const [input, setInput] = React.useState(restored.input);
   479→   479→  const [loading, setLoading] = React.useState(false);
   480→   480→
   481→   481→  const loadingRef = React.useRef(false);
   482→   482→  const idRef = React.useRef(0);
   483→   483→  const taRef = React.useRef<HTMLTextAreaElement>(null);
   484→   484→  const endRef = React.useRef<HTMLDivElement>(null);
   485→   485→  const autoAskedRef = React.useRef<string | null>(null);
   486→   486→
   487→   487→  // persist conversation so section switches keep the thread
   488→   488→  React.useEffect(() => {
   489→   489→    conversationMemory = { turns, input, loading };
   490→   490→  }, [turns, input, loading]);
   491→   491→
   492→   492→  const submit = React.useCallback(
   493→   493→    async (raw: string) => {
   494→   494→      const question = raw.trim();
   495→   495→      if (!question || loadingRef.current) return;
   496→   496→      loadingRef.current = true;
   497→   497→      const id = `turn-${Date.now()}-${idRef.current++}`;
   498→   498→      setTurns((prev) => [...prev, { id, question, response: null, error: null }]);
   499→   499→      setInput('');
   500→   500→      setLoading(true);
   501→   501→      try {
   502→   502→        const res = await fetch('/api/ask', {
   503→   503→          method: 'POST',
   504→   504→          headers: { 'Content-Type': 'application/json' },
   505→   505→          body: JSON.stringify({
   506→   506→            question,
   507→   507→            beginnerMode: useApp.getState().profile?.beginnerMode ?? false,
   508→   508→            madhhab: useApp.getState().profile?.madhhab ?? null,
   509→   509→          }),
   510→   510→        });
   511→   511→        if (!res.ok) {
   512→   512→          const data = (await res.json().catch(() => null)) as { error?: string } | null;
   513→   513→          const message =
   514→   514→            data?.error ?? `BASIRA could not complete the request (status ${res.status}). Please try again.`;
   515→   515→          setTurns((prev) => prev.map((t) => (t.id === id ? { ...t, error: message } : t)));
   516→   516→          toast({ title: 'No answer received', description: message, variant: 'destructive' });
   517→   517→        } else {
   518→   518→          const data = (await res.json()) as AskResponse;
   519→   519→          setTurns((prev) => prev.map((t) => (t.id === id ? { ...t, response: data } : t)));
   520→   520→        }
   521→   521→      } catch {
   522→   522→        const message =
   523→   523→          "Could not reach BASIRA's servers — please check your connection and try again.";
   524→   524→        setTurns((prev) => prev.map((t) => (t.id === id ? { ...t, error: message } : t)));
   525→   525→        toast({ title: 'Connection problem', description: message, variant: 'destructive' });
   526→   526→      } finally {
   527→   527→        loadingRef.current = false;
   528→   528→        setLoading(false);
   529→   529→      }
   530→   530→    },
   531→   531→    [toast]
   532→   532→  );
   533→   533→
   534→   534→  // prefill + auto-submit once when arriving with viewParams.question
   535→   535→  React.useEffect(() => {
   536→   536→    const q = typeof viewParams?.question === 'string' ? viewParams.question.trim() : '';
   537→   537→    if (q && autoAskedRef.current !== q) {
   538→   538→      autoAskedRef.current = q;
   539→   539→      void submit(q);
   540→   540→    }
   541→   541→  }, [viewParams, submit]);
   542→   542→
   543→   543→  // auto-scroll only when the reader is already near the bottom
   544→   544→  React.useEffect(() => {
   545→   545→    if (turns.length === 0) return;
   546→   546→    const nearBottom =
   547→   547→      window.innerHeight + window.scrollY >= document.documentElement.scrollHeight - 320;
   548→   548→    if (!nearBottom) return;
   549→   549→    const t = window.setTimeout(() => {
   550→   550→      endRef.current?.scrollIntoView({ behavior: 'smooth', block: 'nearest' });
   551→   551→    }, 60);
   552→   552→    return () => window.clearTimeout(t);
   553→   553→  }, [turns, loading]);
   554→   554→
   555→   555→  // textarea auto-grow
   556→   556→  React.useEffect(() => {
   557→   557→    const el = taRef.current;
   558→   558→    if (!el) return;
   559→   559→    el.style.height = 'auto';
   560→   560→    el.style.height = `${Math.min(el.scrollHeight, 160)}px`;
   561→   561→  }, [input]);
   562→   562→
   563→   563→  const retry = (turn: Turn) => {
   564→   564→    setTurns((prev) => prev.filter((t) => t.id !== turn.id));
   565→   565→    void submit(turn.question);
   566→   566→  };
   567→   567→
   568→   568→  const startFresh = () => {
   569→   569→    setTurns([]);
   570→   570→    setInput('');
   571→   571→  };
   572→   572→
   573→   573→  const toggleBeginner = (checked: boolean) => {
   574→   574→    void updateProfile({ beginnerMode: checked }).catch((e: unknown) => {
   575→   575→      toast({
   576→   576→        title: 'Could not save preference',
   577→   577→        description: e instanceof Error ? e.message : 'Please try again.',
   578→   578→        variant: 'destructive',
   579→   579→      });
   580→   580→    });
   581→   581→  };
   582→   582→
   583→   583→  const onKeyDown = (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   584→   584→    if (e.key === 'Enter' && !e.shiftKey && !e.nativeEvent.isComposing) {
   585→   585→      e.preventDefault();
   586→   586→      void submit(input);
   587→   587→    }
   588→   588→  };
   589→   589→
   590→   590→  const hasConversation = turns.length > 0 || loading;
   591→   591→
   592→   592→  return (
   593→   593→    <div className="space-y-5">
   594→   594→      {!hasConversation ? (
   595→   595→        <Hero onAsk={(q) => void submit(q)} />
   596→   596→      ) : (
   597→   597→        <>
   598→   598→          <header className="flex items-center justify-between gap-3">
   599→   599→            <div className="flex flex-wrap items-baseline gap-2.5">
   600→   600→              <h1 className="font-display text-xl font-semibold text-foreground">Ask BASIRA</h1>
   601→   601→              <span className="font-arabic text-base text-muted-foreground" lang="ar" dir="rtl">
   602→   602→                اسأل بصيرة
   603→   603→              </span>
   604→   604→            </div>
   605→   605→            {turns.length > 0 && (
   606→   606→              <Button
   607→   607→                type="button"
   608→   608→                variant="ghost"
   609→   609→                size="sm"
   610→   610→                onClick={startFresh}
   611→   611→                className="min-h-11 rounded-lg px-3 text-muted-foreground hover:text-foreground"
   612→   612→              >
   613→   613→                <RotateCcw className="mr-1.5 h-4 w-4" aria-hidden />
   614→   614→                Start fresh
   615→   615→              </Button>
   616→   616→            )}
   617→   617→          </header>
   618→   618→          <div className="ornament-line" aria-hidden />
   619→   619→        </>
   620→   620→      )}
   621→   621→
   622→   622→      {/* ————— conversation ————— */}
   623→   623→      <div className="space-y-6" role="log" aria-label="Conversation with BASIRA" aria-live="polite">
