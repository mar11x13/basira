    88→async function fetchJson<T>(url: string, init?: RequestInit): Promise<T | null> {
    89→  try {
    90→    const res = await fetch(url, { cache: 'no-store', ...init });
    91→    if (!res.ok) return null;
    92→    return (await res.json()) as T;
    93→  } catch {
    94→    return null;
    95→  }
    96→}

    98→/** Honest count: the search API caps returned items, so say "Showing X of Y" when that happens. */
    99→function countLabel(shown: number, total: number): string {

   101→  return `${shown}`;

   145→function SurahBrowser({ progress, onOpen }: { progress: Record<number, number>; onOpen: (n: number) => void }) {
   146→  const [filter, setFilter] = React.useState('');
   147→
   148→  const q = filter.trim().toLowerCase();
   149→  const filtered = React.useMemo(() => {
   150→    if (!q) return SURAHS;
   151→    return SURAHS.filter(
   152→      (s) =>
   153→        String(s.number) === q ||
   154→        s.transliteration.toLowerCase().includes(q) ||
   155→        s.english.toLowerCase().includes(q)
   156→    );
   157→  }, [q]);
   158→
   159→  const startedCount = Object.keys(progress).length;
   160→
   161→  return (
   162→    <section aria-label="Surah browser">
   163→      <div className="relative mb-3">
   164→        <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground pointer-events-none" aria-hidden />
   165→        <Input
   166→          value={filter}
   167→          onChange={(e) => setFilter(e.target.value)}
   168→          type="search"
   169→          placeholder="Filter surahs — number, name, or meaning (e.g. 2, Maryam, light)"
   170→          aria-label="Filter the surah list"
   171→          className="pl-10 h-11 rounded-xl bg-card border-border/80"
   172→        />
   173→      </div>
   174→
   175→      {startedCount > 0 && (
   176→        <p className="text-xs text-muted-foreground mb-2 px-1">
   177→          You have reading progress in {startedCount} {startedCount === 1 ? 'surah' : 'surahs'} — look for the
   178→          <span className="text-primary font-medium"> Continue </span>
   179→          chip.
   180→        </p>
   181→      )}
   182→
   183→      {filtered.length === 0 ? (
   184→        <EmptyState
   185→          icon={<SearchX className="h-10 w-10" aria-hidden />}
   186→          title="No surah matches that filter"

   195→            const lastAyah = progress[s.number];

   197→              <li key={s.number}>
   198→                <button

   249→const STATUS_LABEL: Record<SurahStatus, string> = {
   250→  external: 'Full surah · external dataset',
   251→  cached: 'Full surah · cached locally',
   252→  'local-curated': 'Curated ayahs only',

   254→
   255→function SurahReader({
   256→  surahNumber,
   257→  lastRead,
   258→  onBack,
   259→  onProgress,
   260→}: {
   261→  surahNumber: number;
   262→  lastRead?: number;
   263→  onBack: () => void;
   264→  onProgress: (surahNumber: number, lastAyah: number) => void;
   265→}) {
   266→  const { toast } = useToast();
   267→  const toggleBookmark = useApp((s) => s.toggleBookmark);
   268→  const openRecordBySlug = useApp((s) => s.openRecordBySlug);
   269→  const bookmarkSlugs = useApp((s) => s.bookmarkSlugs);
   270→
   271→  const [data, setData] = React.useState<SurahResponse | null>(null);
   272→  const [state, setState] = React.useState<'loading' | 'ready' | 'error'>('loading');
   273→  const [reloadTick, setReloadTick] = React.useState(0);
   274→  const [marking, setMarking] = React.useState<number | null>(null);
   275→
   276→  React.useEffect(() => {
   277→    let alive = true;
   278→    setState('loading');
   279→    void (async () => {
   280→      const res = await fetchJson<SurahResponse>(`/api/quran/surah/${surahNumber}`);
   281→      if (!alive) return;
   282→      if (res && Array.isArray(res.ayahs)) {
   283→        setData(res);
   284→        setState('ready');
   285→      } else {
   286→        setState('error');
   287→      }
   288→    })();
   289→    return () => {
   290→      alive = false;
   291→    };
   292→  }, [surahNumber, reloadTick]);
   293→
   294→  // After loading, jump to the last-read ayah (or the top of the surah header).
   295→  React.useEffect(() => {
   296→    if (state !== 'ready') return;
   297→    const lastReadEl = typeof lastRead === 'number' ? document.getElementById(`ayah-${surahNumber}-${lastRead}`) : null;
   298→    const target = lastReadEl ?? document.getElementById(`surah-top-${surahNumber}`);
   299→    if (target) {
   300→      requestAnimationFrame(() => {
   301→        target.scrollIntoView({ behavior: 'smooth', block: lastReadEl ? 'center' : 'start' });
   302→      });
   303→    }
   304→  }, [state, surahNumber]);
   305→
   306→  const markLastRead = async (ayah: number) => {
   307→    if (marking !== null) return;
   308→    setMarking(ayah);
   309→    const res = await fetchJson<{ ok: boolean }>('/api/progress', {
   310→      method: 'POST',
   311→      headers: { 'Content-Type': 'application/json' },
   312→      body: JSON.stringify({ kind: 'reading', surahNumber, lastAyah: ayah }),
   313→    });
   314→    setMarking(null);
   315→    if (res?.ok) {
   316→      onProgress(surahNumber, ayah);
   317→      toast({
   318→        title: 'Reading position saved',
   319→        description: `Last read: ${getSurah(surahNumber)?.transliteration ?? `Surah ${surahNumber}`} · ayah ${ayah}.`,
   320→      });
   321→    } else {
   322→      toast({
   323→        title: 'Could not save reading position',
   324→        description: 'Please check your connection and try again.',
   325→      });
   326→    }
   327→  };
   328→
   329→  const surahMeta = data?.surah ?? getSurah(surahNumber);
   330→
   331→  return (
   332→    <section aria-label={`Surah ${surahNumber} reader`}>
   333→      {/* Reader toolbar */}
   334→      <div className="flex items-center justify-between gap-2 mb-4">
   335→        <Button
   336→          variant="ghost"
   337→          onClick={onBack}
   338→          className="h-11 px-3 -ml-2 text-muted-foreground hover:text-foreground"
   339→          aria-label="Back to the surah list"
   340→        >
   341→          <ChevronLeft className="h-5 w-5 mr-1" aria-hidden />
   342→          All surahs
   343→        </Button>
   344→        {typeof lastRead === 'number' && (
   345→          <span className="inline-flex items-center gap-1.5 text-xs text-muted-foreground bg-muted/60 border border-border/60 rounded-full px-3 py-2">
   346→            <BookmarkCheck className="h-3.5 w-3.5 text-primary" aria-hidden />
   347→            Last read: ayah {lastRead}
   348→          </span>
   349→        )}
   350→      </div>
   351→
   352→      {state === 'loading' && (
   353→        <div role="status" aria-busy="true" className="space-y-4">
   354→          <span className="sr-only">Loading surah…</span>
   355→          <Skeleton className="h-44 w-full rounded-xl" />
   356→          <Skeleton className="h-36 w-full rounded-xl" />
   357→          <Skeleton className="h-36 w-full rounded-xl" />
   358→        </div>
   359→      )}
   360→
   361→      {state === 'error' && (
   362→        <ErrorState what={`surah ${surahNumber}`} onRetry={() => setReloadTick((t) => t + 1)} />
   363→      )}
   364→
   365→      {state === 'ready' && data && surahMeta && (
   366→        <>
   367→          {/* Surah header card */}
   368→          <Card className="paper-card overflow-hidden border-border/80 mb-5">
   369→            <div className="pattern-khatim pattern-fade h-1.5 w-full" aria-hidden />
   370→            <CardContent className="p-4 sm:p-6 text-center">
   371→              <p className="text-[0.65rem] uppercase tracking-[0.25em] text-muted-foreground">
   372→                Surah {surahMeta.number} of 114
   373→              </p>
   374→              <h2
   375→                id={`surah-top-${surahNumber}`}
   376→                className="font-arabic text-4xl sm:text-5xl text-foreground leading-[1.7] my-3"
   377→                dir="rtl"
   378→                lang="ar"
   379→              >
   380→                {surahMeta.arabic}
   381→              </h2>
   382→              <p className="font-display text-xl sm:text-2xl font-semibold text-foreground">{surahMeta.transliteration}</p>
   383→              <p className="text-sm text-muted-foreground italic mt-0.5">{surahMeta.english}</p>
   384→              <div className="flex items-center justify-center gap-2 flex-wrap mt-4">
   385→                <Badge variant="outline" className="rounded-full px-3 py-1 text-xs font-medium">
   386→                  {surahMeta.place}
   387→                </Badge>
   388→                <Badge variant="outline" className="rounded-full px-3 py-1 text-xs font-medium">
   389→                  {surahMeta.ayahs} {surahMeta.ayahs === 1 ? 'ayah' : 'ayahs'}
   390→                </Badge>
   391→                <Badge
   392→                  variant="outline"
   393→                  className={cn(
   394→                    'rounded-full px-3 py-1 text-xs font-medium',
   395→                    data.status === 'local-curated' ? 'border-gold/50 text-gold' : 'border-primary/30 text-primary'
   396→                  )}
   397→                >
   398→                  {STATUS_LABEL[data.status]}
   399→                </Badge>
   400→              </div>
   401→              <p className="text-[0.68rem] text-muted-foreground mt-4 leading-relaxed">
   402→                Translation: {data.translationSource}
   403→                {data.source ? ` · ${data.source}` : ''}
   404→              </p>
   405→            </CardContent>
   406→          </Card>
   407→
   408→          {data.status === 'local-curated' && (
   409→            <Alert className="mb-5 border-gold/40 bg-gold/10">
   410→              <CloudOff className="h-4 w-4 text-gold" aria-hidden />
   411→              <AlertTitle className="text-foreground">Only BASIRA&apos;s curated ayahs are available right now</AlertTitle>
   412→              <AlertDescription>{data.note}</AlertDescription>
   413→            </Alert>
   414→          )}
   415→
   416→          {data.ayahs.length === 0 ? (
   417→            <EmptyState
   418→              icon={<BookOpen className="h-10 w-10" aria-hidden />}
   419→              title="No curated ayahs for this surah yet"
   420→              hint="The external Quran dataset could not be reached and BASIRA has not curated this surah yet. Try the Topics tab — every ayah there is verified."
   421→            />
   422→          ) : (
   423→            <div className="space-y-3 sm:space-y-4">
   424→              {data.ayahs.map((a) => {
   425→                const isLastRead = lastRead === a.numberInSurah;
   426→                const bookmarked = a.slug ? bookmarkSlugs.has(a.slug) : false;
   427→                return (
   428→                  <article
   429→                    key={a.numberInSurah}
   430→                    id={`ayah-${surahNumber}-${a.numberInSurah}`}
   431→                    className={cn(
   432→                      'paper-card rounded-xl border p-4 sm:p-5 transition-shadow',
   433→                      isLastRead ? 'border-primary/50 shadow-md' : 'border-border/80 shadow-sm hover:shadow-md'
   434→                    )}
   435→                  >
   436→                    <div className="flex items-center justify-between gap-2">
   437→                      <span
   438→                        className="flex h-8 w-8 shrink-0 items-center justify-center rounded-full bg-primary/10 text-primary text-xs font-bold tabular-nums border border-primary/25"
   439→                        aria-hidden
   440→                      >
   441→                        {a.numberInSurah}
   442→                      </span>
   443→                      <span className="sr-only">Ayah {a.numberInSurah}</span>
   444→                      <Button
   445→                        variant="ghost"
   446→                        size="sm"
   447→                        onClick={() => void markLastRead(a.numberInSurah)}
   448→                        disabled={marking !== null}
   449→                        className={cn(
   450→                          'h-11 px-3 text-xs',
   451→                          isLastRead ? 'text-primary hover:text-primary' : 'text-muted-foreground hover:text-foreground'
   452→                        )}
   453→                        aria-label={`Set ayah ${a.numberInSurah} as your last read position`}
   454→                      >
   455→                        {marking === a.numberInSurah ? (
   456→                          <Loader2 className="h-4 w-4 mr-1.5 animate-spin" aria-hidden />
   457→                        ) : (
   458→                          <BookmarkCheck
   459→                            className={cn('h-4 w-4 mr-1.5', isLastRead && 'fill-primary/20')}
   460→                            aria-hidden
   461→                          />
   462→                        )}
   463→                        {isLastRead ? 'Last read' : 'Set as last read'}
   464→                      </Button>
   465→                    </div>
   466→
   467→                    <ArabicText text={a.arabic} size="lg" />
   468→                    <p className="text-foreground/90 text-[0.95rem] leading-relaxed">{a.translation}</p>
   469→
   470→                    {a.excerptOnly && (
   471→                      <p className="text-xs italic text-muted-foreground mt-2">
   472→                        Excerpt of the ayah, marked with an ellipsis — read the full verse in a printed mushaf.
   473→                      </p>

   475→
   476→                    {a.slug && (
   477→                      <footer className="mt-3 pt-3 border-t border-border/60 flex items-center gap-2 flex-wrap">
   478→                        <Button
   479→                          variant="outline"
   480→                          size="sm"
   481→                          className="h-11 px-4"
   482→                          onClick={() => void toggleBookmark(a.slug!)}
   483→                          aria-pressed={bookmarked}
   484→                          aria-label={bookmarked ? 'Remove this ayah from bookmarks' : 'Bookmark this ayah'}
   485→                        >
   486→                          <Bookmark className={cn('h-4 w-4 mr-1.5', bookmarked && 'fill-current text-gold')} aria-hidden />
   487→                          {bookmarked ? 'Saved' : 'Save'}
   488→                        </Button>
   489→                        <Button
   490→                          variant="ghost"
   491→                          size="sm"
   492→                          className="h-11 px-4 text-muted-foreground hover:text-foreground"
   493→                          onClick={() => void openRecordBySlug(a.slug!)}
   494→                          aria-label="Inspect the verified source record for this ayah"
   495→                        >
   496→                          <BookOpen className="h-4 w-4 mr-1.5" aria-hidden />
   497→                          View source
   498→                        </Button>
   499→                        <span className="ml-auto text-[0.65rem] text-muted-foreground">Curated &amp; verified</span>
   500→                      </footer>

   506→              <div className="text-center py-4">
   507→                <p className="text-xs text-muted-foreground">
   508→                  End of {surahMeta.transliteration} · {surahMeta.ayahs} {surahMeta.ayahs === 1 ? 'ayah' : 'ayahs'}
   509→                </p>
   510→                <div className="ornament-line mt-3" />
   511→              </div>
   512→            </div>
   513→          )}
   514→        </>
   515→      )}
   516→    </section>
   517→  );
   518→}

   525→  const [topic, setTopic] = React.useState<string>('patience');
   526→  const [items, setItems] = React.useState<SourceRecord[]>([]);
   527→  const [total, setTotal] = React.useState(0);
   528→  const [loading, setLoading] = React.useState(true);
   529→  const [failed, setFailed] = React.useState(false);
   530→  const [reloadTick, setReloadTick] = React.useState(0);
   531→
   532→  React.useEffect(() => {
   533→    let alive = true;
   534→    setLoading(true);
   535→    setFailed(false);
   536→    void (async () => {
   537→      const res = await fetchJson<ContentResponse>(
   538→        `/api/content?type=QURAN&topic=${encodeURIComponent(topic)}&limit=60`
   539→      );
   540→      if (!alive) return;
   541→      if (res) {
   542→        setItems(res.items ?? []);
   543→        setTotal(res.total ?? 0);
   544→        setLoading(false);
   545→      } else {
   546→        setFailed(true);
   547→        setLoading(false);
   548→      }
   549→    })();
   550→    return () => {
   551→      alive = false;
   552→    };
   553→  }, [topic, reloadTick]);
   554→
   555→  return (
   556→    <section aria-label="Quran topics">
   557→      <div className="flex flex-wrap gap-2" role="group" aria-label="Choose a topic">
   558→        {QURAN_TOPICS.map((t) => {
   559→          const active = topic === t;
   560→          return (
   561→            <button
   562→              key={t}
   563→              onClick={() => setTopic(t)}
   564→              className={cn(
   565→                'min-h-11 px-4 rounded-full text-sm font-medium border transition-colors focus-ring capitalize',
   566→                active
   567→                  ? 'bg-primary text-primary-foreground border-primary shadow-sm'
   568→                  : 'bg-card text-muted-foreground border-border/80 hover:border-primary/40 hover:text-foreground'
   569→              )}
   570→              aria-pressed={active}
   571→            >
   572→              {t}
   573→            </button>
   574→          );
   575→        })}
   576→      </div>
   577→
   578→      <div className="mt-5">
   579→        <p className="text-sm text-muted-foreground mb-3 capitalize" role="status" aria-live="polite">
   580→          {loading ? (
   581→            'Loading…'
   582→          ) : failed ? null : (
   583→            <>
   584→              <span className="text-foreground font-semibold">{countLabel(items.length, total)}</span> verified{' '}
   585→              {items.length === 1 ? 'ayah' : 'ayahs'} about {topic}
   586→            </>
   587→          )}
   588→        </p>
   589→
   590→        {loading && <LoadingList label={`Loading ayahs about ${topic}…`} />}
   591→        {!loading && failed && <ErrorState what={`ayahs about ${topic}`} onRetry={() => setReloadTick((t) => t + 1)} />}
   592→        {!loading && !failed && items.length === 0 && (
   593→          <EmptyState
   594→            icon={<BookOpen className="h-10 w-10" aria-hidden />}
   595→            title="No curated ayahs for this topic yet"
   596→            hint="BASIRA only shows verified records and has not curated this topic yet — try another topic."
   597→          />
   598→        )}
   599→        {!loading && !failed && items.length > 0 && (
   600→          <div className="space-y-4">
   601→            {items.map((r) => (
   602→              <SourceCard key={r.id} record={r} />
   603→            ))}
   604→          </div>
   605→        )}
   606→      </div>
   607→    </section>
   608→  );
   609→}

   615→function SearchPanel() {

   618→  const [items, setItems] = React.useState<SourceRecord[]>([]);

   620→  const [loading, setLoading] = React.useState(false);

   622→  const [reloadTick, setReloadTick] = React.useState(0);
   623→
   624→  React.useEffect(() => {
   625→    if (!submitted) return;
   626→    let alive = true;
   627→    setLoading(true);
   628→    setFailed(false);
   629→    void (async () => {
   630→      const res = await fetchJson<ContentResponse>(
   631→        `/api/content?type=QURAN&q=${encodeURIComponent(submitted)}&limit=60`
   632→      );
   633→      if (!alive) return;
   634→      if (res) {

   636→        setTotal(res.total ?? 0);

   639→        setFailed(true);
   640→        setLoading(false);

   645→    };

   669→          className="h-11 px-5"
   670→          aria-label="Search"
   671→        >
   672→          {loading ? <Loader2 className="h-4 w-4 animate-spin" aria-hidden /> : <Search className="h-4 w-4" aria-hidden />}

   674→        </Button>
   675→      </div>
   676→
   677→      {!query.trim() && (
   678→        <div className="mt-4">
   679→          <p className="text-xs text-muted-foreground mb-2">Try searching for:</p>
   680→          <div className="flex flex-wrap gap-2" role="group" aria-label="Suggested searches">

   685→                  setQuery(s);

   736→export function QuranView() {
   737→  const viewParams = useApp((s) => s.viewParams);
   738→  const [openSurah, setOpenSurah] = React.useState<number | null>(null);
   739→  const [progress, setProgress] = React.useState<Record<number, number>>({});
   740→
   741→  // Reading progress ("Continue reading") — fetched once on mount.
   742→  React.useEffect(() => {
   743→    let alive = true;
   744→    void (async () => {
   745→      const data = await fetchJson<{ reading: { surahNumber: number; lastAyah: number }[] }>('/api/progress');
   746→      if (!alive || !data?.reading) return;
   747→      const map: Record<number, number> = {};
   748→      for (const r of data.reading) {
   749→        if (typeof r?.surahNumber === 'number' && typeof r?.lastAyah === 'number') {
   750→          map[r.surahNumber] = r.lastAyah;
   751→        }
   752→      }
   753→      setProgress(map);
   754→    })();
   755→    return () => {
   756→      alive = false;
   757→    };
   758→  }, []);
   759→
   760→  // Deep-link from Home ("Continue reading") → open the reader directly.
   761→  const surahParam = typeof viewParams.surah === 'number' ? viewParams.surah : undefined;
   762→  React.useEffect(() => {
   763→    if (typeof surahParam === 'number' && surahParam >= 1 && surahParam <= 114) {
   764→      setOpenSurah(surahParam);
   765→    }
   766→  }, [surahParam]);
   767→
   768→  const recordProgress = (surahNumber: number, lastAyah: number) => {
   769→    setProgress((p) => ({ ...p, [surahNumber]: lastAyah }));
   770→  };
   771→
   772→  return (
   773→    <div>
   774→      <ViewHeader
   775→        title="Qur'an"
   776→        titleAr="القرآن الكريم"
   777→        description="Read the full Qur'an with translation, explore verified ayahs by topic, or search BASIRA's checked collection — every result shows its source."
   778→        back
   779→      />
   780→
   781→      <Tabs defaultValue="read">
   782→        <TabsList className="h-11 grid grid-cols-3 w-full max-w-md rounded-xl" aria-label="Quran sections">
   783→          <TabsTrigger value="read" className="text-sm">
   784→            <BookOpen className="h-4 w-4" aria-hidden />
   785→            Read
   786→          </TabsTrigger>
   787→          <TabsTrigger value="topics" className="text-sm">
   788→            <Tags className="h-4 w-4" aria-hidden />
   789→            Topics
   790→          </TabsTrigger>
   791→          <TabsTrigger value="search" className="text-sm">
   792→            <Search className="h-4 w-4" aria-hidden />
   793→            Search
   794→          </TabsTrigger>
   795→        </TabsList>
   796→
   797→        <TabsContent value="read" className="mt-4">
   798→          {openSurah === null ? (
   799→            <SurahBrowser progress={progress} onOpen={(n) => setOpenSurah(n)} />
   800→          ) : (
   801→            <SurahReader
   802→              key={openSurah}
   803→              surahNumber={openSurah}
   804→              lastRead={progress[openSurah]}
   805→              onBack={() => setOpenSurah(null)}
   806→              onProgress={recordProgress}
   807→            />
   808→          )}
   809→        </TabsContent>
   810→
   811→        <TabsContent value="topics" className="mt-4">
   812→          <TopicPanel />
   813→        </TabsContent>
   814→
   815→        <TabsContent value="search" className="mt-4">
   816→          <SearchPanel />
   817→        </TabsContent>
   818→      </Tabs>
   819→    </div>
   820→  );
   821→}
